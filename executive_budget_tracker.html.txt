                    <div id="subwin_${expense.id}" class="sub-window px-4 py-3.5 ${expense.isOpen ? 'block' : 'hidden'}">
                        <div class="flex flex-wrap items-center justify-between gap-2 mb-3 pb-2 border-b border-slate-200/70">
                            <div class="flex items-center gap-2">
                                <span class="text-xs font-extrabold uppercase tracking-wider text-slate-700">📅 Date-Wise Breakdown Window</span>
                                <span class="text-xs font-semibold text-slate-500">(Logged Sum: <strong class="text-rose-600">${formatPKR(subTotal)} PKR</strong>)</span>

                                ↻ Match Category Budget to Logged Sum
                            </button>

                    let val = parseInt(e.target.value);
                    if (isNaN(val) || val < 0) val = 0;

                    numInput.value = val;
                    rangeInput.value = val;

                    const idx = state.expenses.findIndex(item => item.id === expense.id);
                    if (idx !== -1) {
                        state.expenses[idx].amount = val;
                        updateDashboard();
                        saveState();
                    }
                };

                numInput.addEventListener('input', handleUpdate);
                rangeInput.addEventListener('input', handleUpdate);
            });

        // Sub-Window Actions
        window.toggleSubWindow = function(expenseId) {
            const exp = state.expenses.find(e => e.id === expenseId);
            if (exp) {
                exp.isOpen = !exp.isOpen;
                renderList();
                saveState();
            }
        };

        window.addSubItem = function(expenseId) {
            const exp = state.expenses.find(e => e.id === expenseId);
            if (!exp) return;

            const dateEl = document.getElementById(`subDate_${expenseId}`);
            const titleEl = document.getElementById(`subTitle_${expenseId}`);
            const amtEl = document.getElementById(`subAmt_${expenseId}`);

            const title = titleEl.value.trim() || 'Logged Expense';
            const date = dateEl.value || todayStr();
            const amount = Math.max(0, parseInt(amtEl.value) || 0);

            if (!exp.subItems) exp.subItems = [];
            exp.subItems.push({
                id: Date.now(),
                title,
                date,
                amount
            });

            // Auto-sync parent category amount to match the new sum of logged items
            exp.amount = exp.subItems.reduce((sum, item) => sum + (parseInt(item.amount) || 0), 0);

            renderList();
            updateDashboard();
            saveState();
        };

        window.updateSubItem = function(expenseId, subId, field, value) {
            const exp = state.expenses.find(e => e.id === expenseId);
            if (!exp || !exp.subItems) return;
            const sub = exp.subItems.find(s => s.id === subId);
            if (!sub) return;

            if (field === 'amount') {
                sub.amount = Math.max(0, parseInt(value) || 0);
            } else {
                sub[field] = value;
            }

            renderList();
            updateDashboard();
            saveState();
        };

        window.deleteSubItem = function(expenseId, subId) {
            const exp = state.expenses.find(e => e.id === expenseId);
            if (!exp || !exp.subItems) return;
            exp.subItems = exp.subItems.filter(s => s.id !== subId);
            renderList();
            updateDashboard();
            saveState();
        };

        window.syncTotalFromSubs = function(expenseId) {
            const exp = state.expenses.find(e => e.id === expenseId);
            if (!exp) return;
            const sum = (exp.subItems || []).reduce((acc, s) => acc + (parseInt(s.amount) || 0), 0);
            exp.amount = sum;
            renderList();
            updateDashboard();
            saveState();
        };

        function updateDashboard() {
            const total = state.expenses.reduce((sum, item) => sum + (parseInt(item.amount) || 0), 0);
            const remaining = state.salary - total;
            const pct = state.salary > 0 ? Math.min(100, Math.round((total / state.salary) * 100)) : 0;
            const rawPct = state.salary > 0 ? ((total / state.salary) * 100).toFixed(1) : 0;

            totalEl.innerText = formatPKR(total);
            remainingEl.innerText = formatPKR(remaining);
            utilizationEl.innerText = `${rawPct}%`;
            progressBar.style.width = `${pct}%`;

            const totalSubLogs = state.expenses.reduce((sum, e) => sum + ((e.subItems || []).length), 0);
            activeSubItemsEl.innerText = `${totalSubLogs} Logged Entries`;

            if (remaining < 0) {
                remainingEl.classList.remove('text-emerald-600');
                remainingEl.classList.add('text-rose-600');
                warningMsg.classList.remove('hidden');
            } else {
                remainingEl.classList.remove('text-rose-600');
                remainingEl.classList.add('text-emerald-600');
                warningMsg.classList.add('hidden');
            }

            if (donutChart) {
                const activeExpenses = state.expenses.filter(e => e.amount > 0);
                if (activeExpenses.length === 0) {
                    donutChart.data.labels = ['Unallocated Salary'];
                    donutChart.data.datasets[0].data = [state.salary || 1];
                    donutChart.data.datasets[0].backgroundColor = ['#cbd5e1'];
                } else {
                    const labels = activeExpenses.map(e => e.name);
                    const data = activeExpenses.map(e => e.amount);
                    const bgColors = activeExpenses.map(e => e.color);

                    if (remaining > 0) {
                        labels.push('Remaining / Unallocated');
                        data.push(remaining);
                        bgColors.push('rgba(16, 185, 129, 0.28)');
                    }

                    donutChart.data.labels = labels;
                    donutChart.data.datasets[0].data = data;
                    donutChart.data.datasets[0].backgroundColor = bgColors;
                }
                donutChart.update();
            }
        }

        function initChart() {
            const ctx = document.getElementById('budgetChart').getContext('2d');
            donutChart = new Chart(ctx, {
                type: 'doughnut',
                data: {
                    labels: [],
                    datasets: [{
                        data: [],
                        backgroundColor: [],
                        borderWidth: 2,
                        borderColor: 'rgba(255, 255, 255, 0.9)',
                        hoverOffset: 12,
                        borderRadius: 6
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    cutout: '68%',
                    plugins: {
                        legend: { display: false },
                        tooltip: {
                            backgroundColor: 'rgba(15, 23, 42, 0.92)',
                            titleColor: '#fff',
                            bodyColor: '#f8fafc',
                            padding: 12,
                            cornerRadius: 12,
                            callbacks: {
                                label: function(context) {
                                    const val = context.raw || 0;
                                    const pct = state.salary > 0 ? ((val / state.salary) * 100).toFixed(1) : 0;
                                    return ` ${context.label}: Rs. ${formatPKR(val)} (${pct}%)`;
                                }
                            }
                        }
                    }
                }
            });
        }

        // Add New Master Category
        document.getElementById('addCatBtn').addEventListener('click', () => {
            const nameInput = document.getElementById('newCatName');
            const amountInput = document.getElementById('newCatAmount');
            const errorMsg = document.getElementById('addError');
            
            const name = nameInput.value.trim();
            let amount = parseInt(amountInput.value);
            if (isNaN(amount) || amount < 0) amount = 0;

            if (!name) {
                errorMsg.innerText = "Please enter a category title.";
                errorMsg.classList.remove('hidden');
                return;
            }

            errorMsg.classList.add('hidden');

            const newColor = colors[state.nextColorIndex % colors.length];
            state.nextColorIndex++;

            state.expenses.push({
                id: Date.now(),
                name: name,
                amount: amount,
                color: newColor,
                isOpen: true,
                subItems: amount > 0 ? [{ id: Date.now() + 1, title: `Initial ${name} Allocation`, date: todayStr(), amount: amount }] : []
            });

            nameInput.value = '';
            amountInput.value = '';

            renderList();
            updateDashboard();
            saveState();

            setTimeout(() => {
                listContainer.scrollTop = listContainer.scrollHeight;
            }, 100);
        });

        window.deleteExpense = function(id) {
            state.expenses = state.expenses.filter(e => e.id !== id);
            renderList();
            updateDashboard();
            saveState();
        };

        // Executive ASCII Report Generator
        window.openReportModal = function() {
            const total = state.expenses.reduce((s, i) => s + (parseInt(i.amount) || 0), 0);
            const rem = state.salary - total;
            let lines = [];
            lines.push("======================================================================");
            lines.push(`                 ${(state.heading || "EXECUTIVE MONTHLY BUDGET").toUpperCase()}            `);
            lines.push("======================================================================");
            lines.push(` MONTHLY SALARY : Rs. ${formatPKR(state.salary).padEnd(15)} TOTAL ALLOCATED : Rs. ${formatPKR(total)}`);
            lines.push(` REMAINING BAL  : Rs. ${formatPKR(rem).padEnd(15)} GENERATED DATE  : ${todayStr()}`);
            lines.push("----------------------------------------------------------------------");
            lines.push(" CATEGORY                         BUDGET (PKR)    SHARE (%)   ENTRIES ");
            lines.push("----------------------------------------------------------------------");

            state.expenses.forEach(e => {
                const pct = state.salary > 0 ? ((e.amount / state.salary) * 100).toFixed(1) + '%' : '0.0%';
                const nameCol = e.name.substring(0, 30).padEnd(32);
                const amtCol = formatPKR(e.amount).padStart(12);
                const pctCol = pct.padStart(11);
                const cntCol = String((e.subItems || []).length).padStart(9);
                lines.push(` ${nameCol} ${amtCol} ${pctCol} ${cntCol}`);

                if (e.subItems && e.subItems.length > 0) {
                    e.subItems.forEach(sub => {
                        const subDesc = `[${sub.date}] ${sub.title}`.substring(0, 42).padEnd(44);
                        const subAmt = (`Rs. ${formatPKR(sub.amount)}`).padStart(14);
                        lines.push(`    └─ ${subDesc} ${subAmt}`);
                    });
                }
            });
            lines.push("======================================================================");

            document.getElementById('asciiReportOutput').innerText = lines.join('
');
            document.getElementById('reportModal').classList.remove('hidden');
        };

        window.closeReportModal = function() {
            document.getElementById('reportModal').classList.add('hidden');
        };

        window.copyAsciiReport = function() {
            const text = document.getElementById('asciiReportOutput').innerText;
            navigator.clipboard.writeText(text);
            alert('Executive ASCII Report copied to clipboard!');
        };

        window.onload = init;
